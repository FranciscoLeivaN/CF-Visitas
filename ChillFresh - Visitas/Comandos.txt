// Segundo paso - Crea server.js con este contenido //

// Inicio de contenido //
import express from 'express';
import cors from 'cors';
import bodyParser from 'body-parser';

const app = express();
const PORT = 5000;

app.use(cors());
app.use(bodyParser.json());

app.get('/api', (req, res) => {
  res.json({ message: 'Hello from Node.js backend!' });
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
// Fin del contenido //

// Modifica vite.config.js: //

// Inicio del contenido //
import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'

export default defineConfig({
  plugins: [react()],
  server: {
    proxy: {
      '/api': {
        target: 'http://localhost:5000',
        changeOrigin: true,
        secure: false
      }
    }
  }
})
// Fin del contenido //

// Septima parte - Crear conexión frontend-backend //

// Crea un servicio API para llamar al backend desde React //
export const fetchData = async () => {
  const response = await fetch('/api');
  if (!response.ok) {
    throw new Error('Network response was not ok');
  }
  return response.json();
};
// Fin de creación de servicio //

// Octava parte - Modifica src/App.jsx para probar la conexión //

// Inicio de contenido //
import { useEffect, useState } from 'react';
import { fetchData } from './api';
import './App.css';

function App() {
  const [message, setMessage] = useState('');

  useEffect(() => {
    fetchData()
      .then(data => setMessage(data.message))
      .catch(error => console.error('Error:', error));
  }, []);

  return (
    <div className="App">
      <h1>Fullstack Vite + React + Node.js</h1>
      <p>Backend dice: {message}</p>
    </div>
  );
}

export default App;
// Fin de contenido //
